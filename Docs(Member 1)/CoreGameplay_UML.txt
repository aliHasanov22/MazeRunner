          - Visibility: + (public), # (protected), - (private)
          - {static} marks static fields/methods; {final} marks constants
          - Relationships:
              --|>   Inheritance (extends)            [solid line + hollow triangle]
              ..|>   Interface Realization (implements)[dashed line + hollow triangle]
              *--    Composition (owns lifecycle)     [solid line + filled diamond]
              o--    Aggregation (has-a reference)    [solid line + hollow diamond]
              -->    Directed Association             [solid line + open arrow]
              ..>    Dependency (uses / creates)      [dashed line + open arrow]
              +--    Nested Type (enclosed in parent)
